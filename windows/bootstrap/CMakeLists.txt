cmake_minimum_required(VERSION 3.16)
project(mc_download_bootstrap LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MCD_BUILD_FIXTURE)
  add_executable(portable_fixture WIN32 fixture.cpp)
  set(MCD_TARGET portable_fixture)
else()
  foreach(REQUIRED MCD_CAB_PATH MCD_MANIFEST_PATH MCD_PAYLOAD_ID)
    if(NOT DEFINED ${REQUIRED})
      message(FATAL_ERROR "Missing ${REQUIRED}")
    endif()
  endforeach()
  configure_file(bootstrap.rc.in bootstrap.rc @ONLY)
  configure_file(config.h.in config.h @ONLY)
  add_executable(mc_download_portable WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/bootstrap.rc")
  target_include_directories(mc_download_portable PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(mc_download_portable PRIVATE cabinet bcrypt)
  set(MCD_TARGET mc_download_portable)
endif()
target_compile_features(${MCD_TARGET} PRIVATE cxx_std_17)
target_compile_definitions(${MCD_TARGET} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(${MCD_TARGET} PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(${MCD_TARGET} PRIVATE shell32)
target_link_options(${MCD_TARGET} PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/portable.manifest")
