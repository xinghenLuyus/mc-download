cmake_minimum_required(VERSION 3.16)
project(mc_download_proxy_tests LANGUAGES CXX)
# Standalone test library; production compiles the same code into the runner.
add_library(mc_download_proxy SHARED winhttp_proxy.cpp)
target_compile_definitions(mc_download_proxy PRIVATE MCD_PROXY_BUILD UNICODE _UNICODE)
target_link_libraries(mc_download_proxy PRIVATE winhttp)
target_compile_features(mc_download_proxy PRIVATE cxx_std_17)
set_property(TARGET mc_download_proxy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(mc_download_proxy PRIVATE /W4 /WX /EHsc)
add_executable(mc_download_proxy_test proxy_test.cpp)
target_compile_features(mc_download_proxy_test PRIVATE cxx_std_17)
target_compile_options(mc_download_proxy_test PRIVATE /W4 /WX /EHsc)
set_property(TARGET mc_download_proxy_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(mc_download_proxy_test PRIVATE mc_download_proxy)
